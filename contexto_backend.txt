create or replace function public.actualizar_perfil_propio(
  p_nombre text,
  p_telefono text,
  p_descripcion text,
  p_avatar_path text
)
returns void
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_uid uuid := (select auth.uid());
  v_nombre text := nullif(btrim(p_nombre), '');
  v_telefono text := nullif(btrim(p_telefono), '');
  v_descripcion text := nullif(btrim(p_descripcion), '');
begin
  if v_uid is null or public.don_ricardo_role() is null then
    raise exception 'account_not_active' using errcode = '42501';
  end if;
  if v_nombre is null or char_length(v_nombre) > 120 then
    raise exception 'invalid_name' using errcode = '22023';
  end if;
  if v_telefono is not null and (
    char_length(v_telefono) not between 7 and 25
    or v_telefono !~ '^[+0-9() -]+$'
    or char_length(regexp_replace(v_telefono, '[^0-9]', '', 'g')) not between 7 and 15
  ) then
    raise exception 'invalid_phone' using errcode = '22023';
  end if;
  if v_descripcion is not null and char_length(v_descripcion) > 280 then
    raise exception 'invalid_description' using errcode = '22023';
  end if;
  if p_avatar_path is not null and (
    p_avatar_path !~ ('^' || v_uid::text || '/[0-9a-f-]{36}\.(jpg|png|webp)$')
    or not exists (
      select 1 from storage.objects o
      where o.bucket_id = 'user-avatars'
        and o.name = p_avatar_path
        and o.owner_id = v_uid::text
        and o.metadata->>'mimetype' in ('image/jpeg', 'image/png', 'image/webp')
        and (o.metadata->>'size')::bigint <= 2097152
    )
  ) then
    raise exception 'invalid_avatar' using errcode = '22023';
  end if;

  update public.usuarios
  set nombre = v_nombre,
      telefono = v_telefono,
      descripcion = v_descripcion,
      avatar_path = p_avatar_path
  where id = v_uid and is_active;
  if not found then
    raise exception 'account_not_active' using errcode = '42501';
  end if;
end;
$$;

create or replace function public.gerencia_dashboard(
  p_desde date default null,
  p_estado text default null
)
returns jsonb
language plpgsql
stable
security definer
set search_path = ''
as $$
begin
  if (select auth.uid()) is null or public.don_ricardo_role() is distinct from 'ADMIN' then
    raise exception 'admin_required' using errcode = '42501';
  end if;
  if p_estado is not null and p_estado not in (
    'EN_CAMPO', 'EN_PACKING', 'PENDIENTE_FIRMA', 'APROBADO', 'RECHAZADO'
  ) then
    raise exception 'invalid_status' using errcode = '22023';
  end if;

  return (
    with filtrados as (
      select l.id, l.fundo, l.cultivo, l.agroquimico, l.estado::text as estado,
             l.fecha_cosecha, l.created_at
      from public.lotes l
      where (p_desde is null or l.created_at >= p_desde::timestamptz)
        and l.created_at <= now()
        and (p_estado is null or l.estado::text = p_estado)
    )
    select jsonb_build_object(
      'lotes',
      coalesce(
        (select jsonb_agg(to_jsonb(f) order by f.created_at desc) from filtrados f),
        '[]'::jsonb
      ),
      'documentos',
      (select count(*) from public.documentos d
       where exists (select 1 from filtrados f where f.id = d.lote_id))
    )
  );
end;
$$;

revoke all on function public.actualizar_perfil_propio(text, text, text, text) from public, anon;
revoke all on function public.gerencia_dashboard(date, text) from public, anon;
grant execute on function public.actualizar_perfil_propio(text, text, text, text) to authenticated;
grant execute on function public.gerencia_dashboard(date, text) to authenticated;

commit;

begin;

create or replace function public.admin_set_usuario_rol(
  p_user_id uuid,
  p_rol text
)
returns public.usuarios
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_admin_id uuid := (select auth.uid());
  v_target public.usuarios%rowtype;
  v_admin_count integer;
begin
  if v_admin_id is null then
    raise exception 'auth_required' using errcode = '42501';
  end if;

  if (select public.don_ricardo_role()) is distinct from 'ADMIN' then
    raise exception 'admin_required' using errcode = '42501';
  end if;

  if p_rol not in ('OPERADOR', 'SUPERVISOR', 'ADMIN') then
    raise exception 'invalid_role' using errcode = '22023';
  end if;

  select *
    into v_target
    from public.usuarios
    where id = p_user_id
    for update;

  if v_target.id is null then
    raise exception 'user_not_found' using errcode = '02000';
  end if;

  if p_user_id = v_admin_id and p_rol <> 'ADMIN' then
    raise exception 'cannot_change_own_admin_role' using errcode = '23514';
  end if;

  select count(*)
    into v_admin_count
    from public.usuarios
    where rol = 'ADMIN'::public.user_role
      and is_active
      and id <> p_user_id;

  if v_target.rol = 'ADMIN'::public.user_role and p_rol <> 'ADMIN' and v_admin_count = 0 then
    raise exception 'last_admin_required' using errcode = '23514';
  end if;

  update public.usuarios
  set rol = p_rol::public.user_role
  where id = p_user_id
  returning * into v_target;

  return v_target;
end;
$$;

create or replace function public.admin_set_usuario_estado(
  p_user_id uuid,
  p_is_active boolean
)
returns public.usuarios
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_admin_id uuid := (select auth.uid());
  v_target public.usuarios%rowtype;
  v_admin_count integer;
begin
  if v_admin_id is null then
    raise exception 'auth_required' using errcode = '42501';
  end if;

  if (select public.don_ricardo_role()) is distinct from 'ADMIN' then
    raise exception 'admin_required' using errcode = '42501';
  end if;

  select *
    into v_target
    from public.usuarios
    where id = p_user_id
    for update;

  if v_target.id is null then
    raise exception 'user_not_found' using errcode = '02000';
  end if;

  if p_user_id = v_admin_id and not p_is_active then
    raise exception 'cannot_disable_self' using errcode = '23514';
  end if;

  if v_target.rol = 'ADMIN'::public.user_role and not p_is_active then
    select count(*)
      into v_admin_count
      from public.usuarios
      where rol = 'ADMIN'::public.user_role
        and is_active
        and id <> p_user_id;

    if v_admin_count = 0 then
      raise exception 'last_admin_required' using errcode = '23514';
    end if;
  end if;

  update public.usuarios
  set is_active = p_is_active
  where id = p_user_id
  returning * into v_target;

  return v_target;
end;
$$;

language plpgsql
security definer

as $$
declare
  v_uid uuid := (select auth.uid());
  v_despacho public.despachos%rowtype;
  v_lote public.lotes%rowtype;
  v_motivo text := nullif(btrim(p_motivo), '');
  v_ppm_limit numeric;
begin
  if v_uid is null or (select public.don_ricardo_role()) is distinct from 'ADMIN' then
    raise exception 'admin_required' using errcode = '42501';
  end if;
  if p_aprobar is null then
    raise exception 'decision_required' using errcode = '22023';
  end if;
  if not p_aprobar and (v_motivo is null or char_length(v_motivo) < 5 or char_length(v_motivo) > 500) then
    raise exception 'rejection_reason_required' using errcode = '22023';
  end if;

  select * into v_despacho
  from public.despachos
  where id = p_despacho_id
  for update;
  if v_despacho.id is null then
    raise exception 'dispatch_not_found' using errcode = '02000';
  end if;
  if v_despacho.estado <> 'PENDIENTE_FIRMA' then
    raise exception 'dispatch_already_reviewed' using errcode = '23514';
  end if;
  if not exists (select 1 from public.despacho_documentos d where d.despacho_id = v_despacho.id) then
    raise exception 'dispatch_document_required' using errcode = '23514';
  end if;

  select * into v_lote from public.lotes where id = v_despacho.lote_id for update;
  v_ppm_limit := 0.5;
  if v_despacho.destino = 'UE' then
    v_ppm_limit := 0.3;
  end if;
  if p_aprobar and (
    v_lote.dias_transcurridos < v_lote.dias_carencia
    or v_lote.ppm > v_ppm_limit
  ) then
    raise exception 'lot_fails_destination_requirements' using errcode = '23514';
  end if;

  update public.despachos
  set estado = case when p_aprobar then 'APROBADO' else 'RECHAZADO' end,
      revisado_por = v_uid,
      revisado_at = now(),
      motivo_revision = v_motivo
  where id = v_despacho.id
  returning * into v_despacho;

  if p_aprobar then
    update public.lotes set estado = 'APROBADO' where id = v_lote.id;
  else
    update public.lotes set estado = 'EN_PACKING' where id = v_lote.id;
  end if;
  return v_despacho;
end;
$$;

    const origin = req.headers.get("origin");

    if (!url || !anon || !key || !/^[a-z0-9.-]+$/.test(model)) {
      return reply(503, { error: "El asistente aún no está configurado. Contacta al administrador." });

        if (done) break;
        size += value.byteLength;

        text += decoder.decode(value, { stream: true });

      text += decoder.decode();
